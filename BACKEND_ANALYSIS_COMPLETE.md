const updateData: any = { ...dto };

// Convert batch IDs if provided
if (dto.batches && Array.isArray(dto.batches) && dto.batches.length > 0) {

**Update Strategy**: Full array replacement
- When `batches` or `previousEducation` arrays are provided, they completely replace the existing arrays
- Frontend must send the COMPLETE array including existing items to preserve them
- Missing items will be removed from the database

**Important for Frontend**: Must fetch existing data and merge with updates before submission

export async function updateMyProfile(data: StudentProfileFormData) {
  return withServerActionErrorHandler(async () => {
    const apiUrl = await getApiDomain();
    const httpClient = createHttpClient({ baseURL: apiUrl });

    const result = await httpClient.request('/cpms/students/my-profile', {
      method: 'PATCH',  // ✅ Use PATCH, not PUT
      body: data,
      withAuth: true,
      tokenStrategy: 'auto',
    });

    return {
      success: true,
      data: result.data,
    };
  }, {
    operation: 'update-student-profile',
